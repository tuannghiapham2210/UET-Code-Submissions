-- PHẦN A: THIẾT KẾ CƠ SỞ DỮ LIỆU VÀ NHẬP DỮ LIỆU MẪU

DROP DATABASE IF EXISTS QLDT;
CREATE DATABASE QLDT;
USE QLDT;

-- 1. Bảng SinhVien
CREATE TABLE SinhVien (
    MaSV VARCHAR(10) PRIMARY KEY,
    HoTen VARCHAR(50) NOT NULL,
    NgaySinh DATE,
    DiemTB DOUBLE DEFAULT NULL,
    XepLoai VARCHAR(20) DEFAULT NULL
);

-- 2. Bảng GiaoVien
CREATE TABLE GiaoVien (
    MaGV VARCHAR(10) PRIMARY KEY,
    HoTen VARCHAR(50) NOT NULL,
    Khoa VARCHAR(50)
);

-- 3. Bảng MonHoc
CREATE TABLE MonHoc (
    MaMH VARCHAR(10) PRIMARY KEY,
    TenMH VARCHAR(100) NOT NULL,
    SoTinChi INT NOT NULL CHECK (SoTinChi > 0)
);

-- 4. Bảng GiangDay
CREATE TABLE GiangDay (
    MaGV VARCHAR(10),
    MaMH VARCHAR(10),
    PRIMARY KEY (MaGV, MaMH),
    FOREIGN KEY (MaGV) REFERENCES GiaoVien(MaGV) ON UPDATE CASCADE ON DELETE CASCADE,
    FOREIGN KEY (MaMH) REFERENCES MonHoc(MaMH) ON UPDATE CASCADE ON DELETE CASCADE
);

-- 5. Bảng KetQua
CREATE TABLE KetQua (
    MaSV VARCHAR(10),
    MaMH VARCHAR(10),
    MaGV VARCHAR(10),
    Diem DOUBLE CHECK (Diem >= 0 AND Diem <= 10),
    PRIMARY KEY (MaSV, MaMH),
    FOREIGN KEY (MaSV) REFERENCES SinhVien(MaSV) ON UPDATE CASCADE ON DELETE CASCADE,
    FOREIGN KEY (MaGV, MaMH) REFERENCES GiangDay(MaGV, MaMH) ON UPDATE CASCADE ON DELETE CASCADE
);

-- PHẦN B: TRUY VẤN VÀ CẬP NHẬT DỮ LIỆU THEO YÊU CẦU

-- Tắt chế độ Safe Update để cho phép UPDATE dữ liệu qua bảng phụ
SET SQL_SAFE_UPDATES = 0;

-- 1. Tính điểm trung bình tích lũy (theo trọng số tín chỉ) và xếp loại
UPDATE SinhVien sv
JOIN (
    SELECT 
        kq.MaSV,
        ROUND(SUM(kq.Diem * mh.SoTinChi) / SUM(mh.SoTinChi), 2) AS DiemTB_Tinh
    FROM KetQua kq
    JOIN MonHoc mh ON kq.MaMH = mh.MaMH
    GROUP BY kq.MaSV
) t ON sv.MaSV = t.MaSV
SET 
    sv.DiemTB = t.DiemTB_Tinh,
    sv.XepLoai = CASE 
        WHEN t.DiemTB_Tinh >= 8.5 THEN 'Xuat sac'
        WHEN t.DiemTB_Tinh >= 7.0 THEN 'Kha'
        WHEN t.DiemTB_Tinh >= 5.0 THEN 'Trung binh'
        ELSE 'Yeu'
    END;

-- Bật lại Safe Update
SET SQL_SAFE_UPDATES = 1;

-- Xem bảng SinhVien sau khi cập nhật
SELECT * FROM SinhVien;

-- 2. Danh sách sinh viên còn nợ môn (điểm < 4) và tổng số tín chỉ chưa đạt
SELECT 
    sv.MaSV, 
    sv.HoTen, 
    sv.DiemTB, 
    sv.XepLoai, 
    SUM(mh.SoTinChi) AS TongTinChiNo
FROM SinhVien sv
JOIN KetQua kq ON sv.MaSV = kq.MaSV
JOIN MonHoc mh ON kq.MaMH = mh.MaMH
WHERE kq.Diem < 4.0
GROUP BY sv.MaSV, sv.HoTen, sv.DiemTB, sv.XepLoai;

-- 3. Danh sách sinh viên có điểm trung bình cao nhất
SELECT MaSV, HoTen, DiemTB, XepLoai
FROM SinhVien
WHERE DiemTB = (SELECT MAX(DiemTB) FROM SinhVien);

-- 4. Thống kê: Số sinh viên và phần trăm theo xếp loại
SELECT 
    XepLoai,
    COUNT(*) AS SoLuongSV,
    ROUND(COUNT(*) * 100.0 / (SELECT COUNT(*) FROM SinhVien), 2) AS PhanTram
FROM SinhVien
GROUP BY XepLoai;
